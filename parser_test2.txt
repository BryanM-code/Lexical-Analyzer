! Test 2 (under 21 lines): the sample program from the project
   description, which converts Fahrenheit into Celsius !

function convert (fahr integer)
{
           return 5 * (fahr -32) / 9;
}

@
     integer    low,  high, step;      !declarations!

@
     get (low, high, step);
     while (low  <=  high )
         {  put (low);
             put (convert (low));
             low =  low  +  step;
         }
